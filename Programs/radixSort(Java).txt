public class radixSort {

    public static void main(String[] args) {

        System.out.println();

        int arr[] = { 23, 409, 56, 125, 985, 7, 98 };

        int maxNum = arr[0];
        for (int i = 0; i < arr.length; i++) {

            if (arr[i] > maxNum) {
                maxNum = arr[i];
            }
        }

        int passes = 0;
        while (maxNum != 0) {
            maxNum /= 10;
            passes++;
        }

        int interArr[] = new int[arr.length];

        for(int i = 1; i <= passes; i++){

            int freqArr[] = new int[10];
    
            for(int j = 0; j < arr.length; j++){
    
                freqArr[(arr[j] % (int)Math.pow(10, i)) / ((int)Math.pow(10, i - 1))]++;
            }
    
            for(int j = 1; j < 10; j++){
                freqArr[j] += freqArr[j - 1];
            }
    
            for(int j = arr.length - 1; j >= 0; j--){
    
                interArr[--freqArr[(arr[j] % (int)Math.pow(10, i)) / ((int)Math.pow(10, i - 1))]] = arr[j];
            }
    
            for(int j = 0; j < arr.length; j++){
                arr[j] = interArr[j];
            }
        }

        for(int i = 0; i < arr.length; i++){
            System.out.print(arr[i] + " ");
        }

        System.out.println("\n");
    }
}
